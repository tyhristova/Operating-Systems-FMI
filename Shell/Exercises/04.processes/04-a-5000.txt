Намерете командите на 10-те най-стари процеси в системата.

ps -eo etimes,cmd --sort=-etimes | head -n 11 | tail -n 10

# -e -> показва всички процеси 
# -о -> задава формат (в случая избира колона) -> etimes - общото време от стартирането на процеса досега в секунди, cmd - командата
# head -n 11 | tail -n 10 -> пропуска headline-а
